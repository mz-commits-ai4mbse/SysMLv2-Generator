            package '02_EngineeringLifecycle' {
                action def <SFB_002> 'Engineering Transformation Lifecycle' {
                    doc
                    /* Control-oriented lifecycle for transformation of source-bound
                     * engineering information into reviewed SysML v2 architecture artifacts.
                     *
                     * The model contains explicit review, semantic, evidence, architecture
                     * and representation rework loops. It does not imply that all projects
                     * execute every branch.
                     */
                    first start;
                    then action manageContext : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Manage Engineering Project and Source Context';
                    then action configureRun : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Configure and Orchestrate Processing Runs';
                    then merge inputAssessmentEntry;
                    then action assessInput : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Assess and Prepare Source Information' {
                        out assessmentReady : Boolean;
                    }
                    then decide assessmentGate;
                    if assessInput.assessmentReady then semanticEntry;
                    else reviewInput;
                    action reviewInput : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Support Human Review and Approval' {
                        out acceptInput : Boolean;
                        out requestInputChanges : Boolean;
                        out rejectInput : Boolean;
                    }
                    then decide inputReviewGate;
                    if reviewInput.requestInputChanges then inputAssessmentEntry;
                    if reviewInput.acceptInput then semanticEntry;
                    if reviewInput.rejectInput then stopRun;
                    merge semanticEntry;
                    then action applySemanticGovernance : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Apply Semantic Governance' {
                        out semanticReady : Boolean;
                    }
                    then decide semanticGate;
                    if applySemanticGovernance.semanticReady then coverageEntry;
                    else generateSemanticCandidates;
                    action generateSemanticCandidates : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Generate and Compare Engineering Candidates';
                    then action reviewSemanticResolution : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Support Human Review and Approval' {
                        out semanticResolved : Boolean;
                        out requestSemanticChanges : Boolean;
                        out rejectSemanticResolution : Boolean;
                    }
                    then decide semanticReviewGate;
                    if reviewSemanticResolution.semanticResolved then semanticEntry;
                    if reviewSemanticResolution.requestSemanticChanges then generateSemanticCandidates;
                    if reviewSemanticResolution.rejectSemanticResolution then stopRun;
                    merge coverageEntry;
                    then action assessEvidenceCoverage : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Assess Coverage and Evidence Sufficiency' {
                        out evidenceSufficient : Boolean;
                    }
                    then decide coverageGate;
                    if assessEvidenceCoverage.evidenceSufficient then candidateEntry;
                    else reviewEvidenceDeficiency;
                    action reviewEvidenceDeficiency : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Support Human Review and Approval' {
                        out requestAdditionalInformation : Boolean;
                        out requestAlternativeInterpretation : Boolean;
                        out acceptKnownEvidenceGap : Boolean;
                        out rejectForInsufficientEvidence : Boolean;
                    }
                    then decide evidenceReviewGate;
                    if reviewEvidenceDeficiency.requestAdditionalInformation then inputAssessmentEntry;
                    if reviewEvidenceDeficiency.requestAlternativeInterpretation then candidateEntry;
                    if reviewEvidenceDeficiency.acceptKnownEvidenceGap then candidateEntry;
                    if reviewEvidenceDeficiency.rejectForInsufficientEvidence then stopRun;
                    merge candidateEntry;
                    then action generateEngineeringCandidates : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Generate and Compare Engineering Candidates';
                    then action reviewEngineeringCandidates : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Support Human Review and Approval' {
                        out approveCandidate : Boolean;
                        out requestCandidateChanges : Boolean;
                        out rejectCandidate : Boolean;
                    }
                    then decide candidateReviewGate;
                    if reviewEngineeringCandidates.approveCandidate then architectureEntry;
                    if reviewEngineeringCandidates.requestCandidateChanges then candidateEntry;
                    if reviewEngineeringCandidates.rejectCandidate then stopRun;
                    merge architectureEntry;
                    then action deriveArchitecture : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Derive and Structure Architecture Information';
                    then action validateArchitecture : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Validate and Integrate Architecture Content' {
                        out architectureAccepted : Boolean;
                        out architectureReworkRequired : Boolean;
                        out semanticReworkRequired : Boolean;
                        out evidenceReworkRequired : Boolean;
                        out ambiguityReworkRequired : Boolean;
                    }
                    then decide architectureValidationGate;
                    if validateArchitecture.architectureAccepted then artifactGenerationEntry;
                    if validateArchitecture.architectureReworkRequired then architectureEntry;
                    if validateArchitecture.semanticReworkRequired then semanticEntry;
                    if validateArchitecture.evidenceReworkRequired then coverageEntry;
                    if validateArchitecture.ambiguityReworkRequired then candidateEntry;
                    merge artifactGenerationEntry;
                    then action generateSysMLArtifacts : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Generate and Validate SysML v2 Architecture Artifacts' {
                        out artifactReadyForReview : Boolean;

                    if generateSysMLArtifacts.notationCorrectionRequired then artifactGenerationEntry;
                    if generateSysMLArtifacts.architectureChangeRequired then architectureEntry;

                        out approveArchitectureArtifact : Boolean;
                        out modelChangeRequired : Boolean;
                        out serializationChangeRequired : Boolean;
                        out rejectArchitectureArtifact : Boolean;
                    }

                    if finalReview.approveArchitectureArtifact then completeRun;
                    if finalReview.modelChangeRequired then architectureEntry;
                    if finalReview.serializationChangeRequired then artifactGenerationEntry;
                    if finalReview.rejectArchitectureArtifact then stopRun;
                    action completeRun : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Configure and Orchestrate Processing Runs';
                    then done;
                    action stopRun : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Configure and Orchestrate Processing Runs';
                    then done;
                }

            package '03_ReviewRecoveryAndSupersession' {
                action def <SFB_003> 'Review Recovery and Supersession' {
                    doc
                    /* Operational control behavior for review outcomes, technical retry,
                     * interrupted publication recovery and material-change supersession.
                     *
                     * Retry reuses the current run only when run bindings remain unchanged.
                     * A material binding change creates an explicit successor run.
                     * Earlier processing evidence remains available.
                     */
                    first start;
                    then action classifyContinuationCondition : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Configure and Orchestrate Processing Runs' {
                        out normalContinuation : Boolean;
                        out technicalFailure : Boolean;
                        out interruptedPublication : Boolean;
                        out reviewChangesRequested : Boolean;
                        out reviewedTargetRejected : Boolean;
                    }
                    then decide continuationConditionGate;
                    if classifyContinuationCondition.normalContinuation then resumeStage;
                    if classifyContinuationCondition.technicalFailure then recordTechnicalFailure;
                    if classifyContinuationCondition.interruptedPublication then validateExistingPublication;
                    if classifyContinuationCondition.reviewChangesRequested then assessChangeMateriality;
                    if classifyContinuationCondition.reviewedTargetRejected then recordReviewRejection;
                    action recordTechnicalFailure : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Maintain Processing Evidence and Traceability';
                    then action assessRetryCompatibility : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Configure and Orchestrate Processing Runs' {
                        out bindingsUnchanged : Boolean;
                        out bindingsChanged : Boolean;
                        out retryNotPermitted : Boolean;

                    then decide retryCompatibilityGate;
                    if assessRetryCompatibility.bindingsUnchanged then createNewAttempt;
                    if assessRetryCompatibility.bindingsChanged then createSuccessorRun;
                    if assessRetryCompatibility.retryNotPermitted then blockRun;
                    action assessChangeMateriality : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Configure and Orchestrate Processing Runs' {
                        out bindingsUnchanged : Boolean;
                        out bindingsChanged : Boolean;
                    }
                    then decide changeMaterialityGate;
                    if assessChangeMateriality.bindingsUnchanged then createNewAttempt;
                    if assessChangeMateriality.bindingsChanged then createSuccessorRun;
                    action createNewAttempt : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Configure and Orchestrate Processing Runs';
                    then action retainAttemptEvidence : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Maintain Processing Evidence and Traceability';
                    then resumeStage;
                    action createSuccessorRun : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Configure and Orchestrate Processing Runs';

                    then resumeStage;
                    action validateExistingPublication : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Maintain Processing Evidence and Traceability' {
                        out exactArtifactValid : Boolean;
                        out publicationHistoryInconsistent : Boolean;
                    }

                    if validateExistingPublication.publicationHistoryInconsistent then blockRun;
                    action completePublicationHistory : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Maintain Processing Evidence and Traceability';
                    then resumeStage;
                    action recordReviewRejection : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Support Human Review and Approval';
                    then action completeWithoutPublication : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Configure and Orchestrate Processing Runs';
                    then done;
                    action blockRun : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Configure and Orchestrate Processing Runs';
                    then action presentRecoveryAttention : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Present Processing Status and History';
                    then done;
                    action resumeStage : '02_System'::'02_Functional'::'00_FunctionDefinitions'::'Configure and Orchestrate Processing Runs';
                    then done;
                }
